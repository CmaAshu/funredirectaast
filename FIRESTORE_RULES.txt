========================================================
  PREPOGY FIRESTORE SECURITY RULES
  Paste this into:
  Firebase Console → Firestore Database → Rules tab
========================================================

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // ── Helper: is the current user an admin? ─────────────────────────────
    function isAdmin() {
      return request.auth != null &&
             get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role == 'admin';
    }

    // ── USERS ─────────────────────────────────────────────────────────────
    // Users can read & write their own doc; admins can read all.
    match /users/{userId} {
      allow read:  if request.auth != null && (request.auth.uid == userId || isAdmin());
      allow write: if request.auth != null && request.auth.uid == userId;
    }

    // ── LEADERBOARD ───────────────────────────────────────────────────────
    // Any signed-in user can read; only owner can CREATE their score (no updates).
    match /leaderboard/{weekDoc}/scores/{userId} {
      allow read:   if request.auth != null;
      allow create: if request.auth != null && request.auth.uid == userId;
      allow update, delete: if false;
    }

    // ── REPORTS (user-submitted question corrections) ─────────────────────
    match /reports/{reportId} {
      allow read:   if isAdmin();
      allow create: if request.auth != null;
      allow update: if isAdmin();
      allow delete: if isAdmin();
    }

    // ── BLOG ─────────────────────────────────────────────────────────────
    match /blog/{postId} {
      allow read:  if true;
      allow write: if isAdmin();
    }

    // ── NOTIFICATIONS ─────────────────────────────────────────────────────
    match /notifications/{notifId} {
      allow read:  if true;
      allow write: if isAdmin();
    }

    // ── QUIZ OVERRIDES (admin-created custom sets) ─────────────────────────
    match /quizOverrides/{paperId}/sets/{setName}/questions/{questionId} {
      allow read:  if request.auth != null;
      allow write: if isAdmin();
    }
  }
}

========================================================
  HOW TO APPLY:
  1. Go to console.firebase.google.com
  2. Select your "prepogy" project
  3. Click "Firestore Database" in the left menu
  4. Click the "Rules" tab at the top
  5. Delete all existing rules
  6. Paste the rules above (starting from rules_version = '2')
  7. Click "Publish"

  WHY LEADERBOARD WASN'T UPDATING:
  The old firebase.js had placeholder keys ("YOUR_API_KEY"), so the app
  was not actually connected to Firebase at all. This has been fixed —
  the real config keys are now in firebase.js.

  Also ensure the Firestore rules above are published. The leaderboard
  uses "create" (not update) — each user can only post their score once
  per week per quiz, which is by design.
